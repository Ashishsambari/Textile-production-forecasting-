import numpy as np
import pandas as pd
import os
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
import xgboost as xgb
from xgboost import plot_importance
from sklearn.metrics import mean_squared_error, r2_score

# Load your dataset
data = pd.read_csv('/content/Book2.csv')

# Check the shape and column names of your data
print("Data Shape:", data.shape)
print("Columns:", data.columns)

# Check for missing values
print("Missing Values:\n", data.isna().sum())

# Perform Label Encoding for categorical data (if necessary)
le = LabelEncoder()
for i in data.columns:
    if data[i].dtype == 'object':
        data[i] = le.fit_transform(data[i])

# Split the data into features (X) and the target variable (y)
X = data.drop('Total_pdn_per_order', axis=1)  # Assuming 'Total_pdn_per_order' is the target variable
y = data['Total_pdn_per_order']

# Store R-squared scores in a list
r2_scores = []

# Perform 10 iterations
for iteration in range(10):
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=iteration * 35)

    # Create an XGBoost regressor
    xgb_reg = xgb.XGBRegressor(objective='reg:squarederror', n_estimators=100)

    # Fit the model on the training data
    xgb_reg.fit(X_train, y_train)

    # Make predictions on the test data
    y_pred = xgb_reg.predict(X_test)

    # Calculate the R-squared score
    r2 = r2_score(y_test, y_pred)
    r2_scores.append(r2)
    print(f"Iteration {iteration + 1}: R-squared Score: {r2:.4f}")

# Calculate and print the mean R-squared score
mean_r2 = np.mean(r2_scores)
print(f"\nMean R-squared Score over 10 iterations: {mean_r2:.4f}")
